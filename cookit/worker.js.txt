// Cook It import helper — paste this whole file into a Cloudflare Worker.
// Needs one secret: ANTHROPIC_API_KEY (Settings → Variables and Secrets).

const APP_CODE = 'cindy';
const MODEL = 'claude-sonnet-5-5';

const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
};
const json = (o, s = 200) => new Response(JSON.stringify(o), { status: s, headers: { ...CORS, 'Content-Type': 'application/json' } });

const RULES = `Use Australian English, metric units (g, kg, ml, L, tsp, tbsp, cups ok for small amounts), °C oven temps, and Australian ingredient names (coriander, capsicum, zucchini, eggplant, spring onion, plain flour, caster sugar, mince).
Return ONLY JSON, no other text, in this shape:
{"title":"","servings":4,"calsPerServing":500,"tags":["2-4 short tags e.g. quick, vegetarian, pasta"],"ingredients":["400 g spaghetti","2 cloves garlic, finely chopped"],"steps":["One clear step per item, short enough to read from across the kitchen. Keep times like 10 minutes."],"notes":""}
Ingredients must start with the amount then unit. calsPerServing is your best estimate from the ingredients.`;

async function claude(env, content) {
  const r = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: { 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01', 'content-type': 'application/json' },
    body: JSON.stringify({ model: MODEL, max_tokens: 4000, messages: [{ role: 'user', content }] }),
  });
  const j = await r.json();
  if (!r.ok) throw new Error((j.error && j.error.message) || 'API error ' + r.status);
  const text = j.content.map(c => c.text || '').join('');
  const m = text.match(/\{[\s\S]*\}/);
  if (!m) throw new Error('Could not read a recipe');
  return JSON.parse(m[0]);
}

async function pageText(url) {
  const r = await fetch(url, { headers: { 'User-Agent': 'Mozilla/5.0 (iPhone; CPU iPhone OS 18_0 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/18.0 Mobile/15E148 Safari/604.1' } });
  if (!r.ok) throw new Error('Site returned ' + r.status);
  const html = await r.text();
  const og = (html.match(/<meta[^>]+property=["']og:image["'][^>]+content=["']([^"']+)/i) || html.match(/<meta[^>]+content=["']([^"']+)["'][^>]+property=["']og:image/i) || [])[1] || '';
  // Most recipe sites include structured recipe data — use it if present
  const ld = [...html.matchAll(/<script[^>]*application\/ld\+json[^>]*>([\s\S]*?)<\/script>/gi)].map(m => m[1]).filter(s => /Recipe/.test(s));
  if (ld.length) return { og, text: ld.join('\n').slice(0, 40000) };
  return { og, text: html.replace(/<script[\s\S]*?<\/script>|<style[\s\S]*?<\/style>/gi, ' ').replace(/<[^>]+>/g, ' ').replace(/\s+/g, ' ').slice(0, 40000) };
}

export default {
  async fetch(req, env) {
    if (req.method === 'OPTIONS') return new Response(null, { headers: CORS });
    if (req.method !== 'POST') return json({ error: 'POST only' }, 405);
    try {
      const b = await req.json();
      if (b.code !== APP_CODE) return json({ error: 'Wrong code' }, 403);

      if (b.estimate) {
        const out = await claude(env, `Estimate calories per serving for this recipe serving ${b.estimate.servings}. Ingredients:\n${b.estimate.ingredients.join('\n')}\nReturn ONLY JSON: {"calsPerServing":number}`);
        return json({ calsPerServing: Math.round(out.calsPerServing) });
      }
      if (b.url) {
        const { og, text } = await pageText(b.url);
        const out = await claude(env, `Extract the recipe from this webpage content. Also include "imageUrl": the full URL of the main photo of the finished dish if one is in the content, else "".\n${RULES}\n\nPAGE:\n${text}`);
        out.imageUrl = out.imageUrl || og || '';
        if (out.imageUrl && !/^https?:/.test(out.imageUrl)) out.imageUrl = new URL(out.imageUrl, b.url).href;
        return json(out);
      }
      if (b.image) {
        const [, type, data] = b.image.match(/^data:(image\/\w+);base64,(.*)$/) || [];
        if (!data) return json({ error: 'Bad image' }, 400);
        return json(await claude(env, [
          { type: 'image', source: { type: 'base64', media_type: type, data } },
          { type: 'text', text: `Read the recipe in this photo (could be a cookbook page, screenshot or handwritten card).
Also include "dishBox": if the image contains a photo of the finished dish, its position as [left, top, right, bottom] fractions of the image width/height (0 to 1); otherwise null.\n${RULES}` },
        ]));
      }
      return json({ error: 'Nothing to import' }, 400);
    } catch (e) {
      return json({ error: e.message }, 500);
    }
  },
};
